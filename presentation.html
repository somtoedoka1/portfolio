Implementation

Error-Resilient Orchestration

LangGraph treats the workflow as a graph—if something goes wrong, it reroutes automatically.

# LangGraph node: try to act, reroute on failure
def execute_gui_action(state):
    vision_context = get_screen_state()
    coordinates = llm_vision_parse(vision_context, state.target_element)

    if coordinates:
        open_interface.click(coordinates.x, coordinates.y)
        return {"next_node": "verify_action"}
    else:
        return {"next_node": "error_correction"}  # scroll, zoom, retry
